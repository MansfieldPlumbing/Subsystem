cmake_minimum_required(VERSION 3.20)
project(DirectPortInference LANGUAGES CXX)

# Match VirtuaCam toolchain settings exactly
add_compile_definitions(_WIN32_WINNT=0x0A00)
set(CMAKE_SYSTEM_VERSION "10.0.22621.0")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

# vcpkg toolchain (same as VirtuaCam)
# cmake -DCMAKE_TOOLCHAIN_FILE=C:/BUILD/vcpkg/scripts/buildsystems/vcpkg.cmake ..
find_package(wil CONFIG REQUIRED)

set(LLAMA_DIR "C:/BUILD/llama.cpp")
set(DIRECTPORT_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(DXIL_DIR "C:/BUILD/native_hlsl")

# --- DirectPort SDK (C transport layer) ---
add_library(DirectPort STATIC
    ${DIRECTPORT_DIR}/directportd3d12.cpp
    ${DIRECTPORT_DIR}/directportd3d11.cpp
)
target_include_directories(DirectPort PUBLIC ${DIRECTPORT_DIR})
target_link_libraries(DirectPort PUBLIC d3d12 d3d11 dxgi advapi32)
target_compile_definitions(DirectPort PRIVATE UNICODE _UNICODE)

# --- Inference backend DLL ---
# Mirrors DirectPortBroker / DirectPortConsumer from VirtuaCam CMakeLists
add_library(DirectPortInference SHARED
    DieWorker.cpp
)
target_include_directories(DirectPortInference PRIVATE
    ${DIRECTPORT_DIR}
    ${LLAMA_DIR}/ggml/include
    ${LLAMA_DIR}/include
)
target_link_libraries(DirectPortInference PRIVATE
    DirectPort
    WIL::WIL
    d3d12 dxgi d3dcompiler advapi32
)
target_compile_definitions(DirectPortInference PRIVATE
    UNICODE _UNICODE
    DXIL_DIR="${DXIL_DIR}"
)
target_precompile_headers(DirectPortInference PRIVATE DieWorker.h)

# and register via ggml_backend_register() at init time.
# See: llama.cpp/ggml/src/ggml-backend.cpp -> ggml_backend_init_best()
